# 1. Посмотреть, что изменилось
git status

# 2. Добавить все изменения в индекс
git add .

# 3. Закоммитить с сообщением
git commit -m "fix: сообщения через supabase, RLS, вернул админа"

# 4. Запушить в текущую ветку
git push


Если ветка ещё не связана с remote (первый push):
bash
git push -u origin main
